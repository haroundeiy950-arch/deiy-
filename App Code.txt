function validate(d: any) {
  const e: any = {};
  const alias = d.alias.trim();
  if (!alias) e.alias = 'Vendor alias is required.';
  else if (alias.length < 3) e.alias = 'Alias must be at least 3 characters.';
  const stall = d.stallCode.trim();
  if (!stall) e.stallCode = 'Stall code is required.';
  else if (!STALL_PATTERN.test(stall)) e.stallCode = 'Use the format MS-B07-023.';
  if (!d.category) e.category = 'Select a category.';
  const phone = d.phone.replace(/\s/g, '');
  if (!phone) e.phone = 'Contact number is required.';
  else if (!PHONE_PATTERN.test(phone)) e.phone = 'Use a Rwanda format such as 0781234567 or +250781234567.';
  if (!d.risk) e.risk = 'Select a risk level.';
  if (!d.consent) e.consent = 'Consent confirmation is required.';
  return e;
}

const Ctx = createContext<any>(null);
function SessionProvider({ children }: any) {
  const [draft, setDraft] = useState<any>(emptyDraft);
  const [records, setRecords] = useState<any[]>([seed]);
  const [resetCount, setResetCount] = useState(0);
  const setField = useCallback((k: string, v: any) => setDraft((d: any) => ({ ...d, [k]: v })), []);
  const resetDraft = useCallback(() => {
    setDraft(emptyDraft);
    setResetCount((c) => c + 1);
  }, []);
  const addRecord = useCallback((r: any) => setRecords((p) => [r, ...p]), []);
  const value = useMemo(
    () => ({ draft, setField, resetDraft, records, addRecord, resetCount }),
    [draft, setField, resetDraft, records, addRecord, resetCount]
  );
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}
const useSession = () => useContext(Ctx);

function Chip({ label, bg, fg }: any) {
  return (
    <View style={[s.chip, { backgroundColor: bg }]}>
      <Text style={[s.chipText, { color: fg }]}>{label}</Text>
    </View>
  );
}

function AppButton({ label, onPress, variant = 'primary', flex }: any) {
  const outline = variant !== 'primary';
  const color = variant === 'danger' ? colors.danger : colors.primary;
  return (
    <Pressable
      onPress={onPress}
      accessibilityRole="button"
      accessibilityLabel={label}
      style={({ pressed }) => [
        s.btn,
        flex && s.btnFlex,
        outline ? { borderColor: color, backgroundColor: colors.card } : { backgroundColor: colors.primary, borderColor: colors.primary },
        pressed && { opacity: 0.8 },
      ]}
    >
      <Text style={[s.btnText, { color: outline ? color : '#FFFFFF' }]}>{label}</Text>
    </Pressable>
  );
}

function PhotoPicker({ uri, onChange }: any) {
  const [notice, setNotice] = useState<any>(null);
  const denied = (what: string, other: string) => {
    setNotice({ error: true, text: `${what} permission was denied. Allow it in Settings, or ${other}.` });
    Alert.alert(`${what} permission needed`, `Enable ${what.toLowerCase()} access in Settings, or ${other}.`, [
      { text: 'Not now', style: 'cancel' },
      { text: 'Open Settings', onPress: () => Linking.openSettings() },
    ]);
  };
  const handle = (res: any) => {
    if (res.canceled || !res.assets || res.assets.length === 0) {
      setNotice({ error: false, text: uri ? 'Selection cancelled. Your current photo was kept.' : 'No photo selected. You can try again.' });
      return;
    }
    onChange(res.assets[0].uri);
    setNotice(null);
  };
  const capture = async () => {
    const p = await ImagePicker.requestCameraPermissionsAsync();
    if (!p.granted) return denied('Camera', 'choose a photo from the gallery');
    handle(await ImagePicker.launchCameraAsync({ mediaTypes: ['images'], quality: 0.6 }));
  };
  const pick = async () => {
    const p = await ImagePicker.requestMediaLibraryPermissionsAsync();
    if (!p.granted) return denied('Photo library', 'take a photo with the camera');
    handle(await ImagePicker.launchImageLibraryAsync({ mediaTypes: ['images'], quality: 0.6 }));
  };
  return (
    <View>
      {uri ? (
        <Image source={{ uri }} style={s.preview} accessibilityLabel="Selected evidence photo preview" />
      ) : (
        <View style={s.placeholder}>
          <Text style={{ color: colors.muted, fontSize: 14 }}>No evidence photo yet</Text>
        </View>
      )}
      <View style={s.rowWrap}>
        <AppButton label={uri ? 'Retake with camera' : 'Take photo'} onPress={capture} flex />
        <AppButton label={uri ? 'Replace from gallery' : 'Choose from gallery'} onPress={pick} variant="outline" flex />
      </View>
      {uri ? (
        <View style={s.rowWrap}>
          <AppButton label="Remove photo" onPress={() => { onChange(null); setNotice({ error: false, text: 'Photo removed.' }); }} variant="danger" flex />
        </View>
      ) : null}
      {notice ? (
        <Text style={[s.notice, notice.error ? { backgroundColor: colors.dangerBg, color: colors.danger } : { backgroundColor: colors.infoBg, color: colors.primary }]}>
          {notice.text}
        </Text>
      ) : null}
    </View>
  );
}

function FormScreen({ navigation }: any) {
  const { draft, setField, resetCount } = useSession();
  const [touched, setTouched] = useState<any>({});
  const [attempted, setAttempted] = useState(false);

  useEffect(() => {
    setTouched({});
    setAttempted(false);
  }, [resetCount]);

  const errors = validate(draft);
  const show = (k: string) => (attempted || touched[k] ? errors[k] : undefined);
  const touch = (k: string) => setTouched((t: any) => ({ ...t, [k]: true }));
  const submit = () => {
    setAttempted(true);
    if (Object.keys(errors).length === 0) navigation.navigate('Review');
  };
  return (
    <ScrollView style={s.screen} contentContainerStyle={{ padding: space.lg, paddingBottom: 48 }} keyboardShouldPersistTaps="handled">
      <Text style={s.note}>Use fictional data only. Do not enter real names or phone numbers.</Text>

      <Field label="Vendor alias" error={show('alias')}>
        <TextInput
          value={draft.alias}
          onChangeText={(v) => setField('alias', v)}
          onBlur={() => touch('alias')}
          placeholder="e.g. Vendor Alpha"
          placeholderTextColor={colors.muted}
          style={[s.input, show('alias') && s.inputErr]}
          accessibilityLabel="Vendor alias"
        />
      </Field>

      <Field label="Stall code" error={show('stallCode')}>
        <TextInput
          value={draft.stallCode}
          onChangeText={(v) => setField('stallCode', v.toUpperCase())}
          onBlur={() => touch('stallCode')}
          placeholder="MS-B07-023"
          placeholderTextColor={colors.muted}
          autoCapitalize="characters"
          autoCorrect={false}
          style={[s.input, show('stallCode') && s.inputErr]}
          accessibilityLabel="Stall code"
        />
      </Field>

      <Field label="Category" error={show('category')}>
        <ChoiceRow options={CATEGORIES} value={draft.category} onChange={(v: string) => { setField('category', v); touch('category'); }} />
      </Field>

      <Field label="Contact number (fictional)" error={show('phone')}>
        <TextInput
          value={draft.phone}
          onChangeText={(v) => setField('phone', v)}
          onBlur={() => touch('phone')}
          placeholder="0781234567"
          placeholderTextColor={colors.muted}
          keyboardType="phone-pad"
          style={[s.input, show('phone') && s.inputErr]}
          accessibilityLabel="Contact number"
        />
      </Field>

      <Field label="Risk level" error={show('risk')}>
        <ChoiceRow options={RISKS} value={draft.risk} onChange={(v: string) => { setField('risk', v); touch('risk'); }} />
      </Field>

      <Field label="Consent" error={show('consent')}>
        <View style={{ flexDirection: 'row', alignItems: 'center' }}>
          <Switch
            value={draft.consent}
            onValueChange={(v) => { setField('consent', v); touch('consent'); }}
            trackColor={{ true: colors.primary, false: colors.border }}
            accessibilityLabel="Consent confirmation"
          />
          <Text style={{ flex: 1, marginLeft: space.md, fontSize: 14, color: colors.text }}>The vendor agreed to this fictional inspection record.</Text>
        </View>
      </Field>

      <Field label="Evidence photo">
        <PhotoPicker uri={draft.imageUri} onChange={(u: any) => setField('imageUri', u)} />
      </Field>

      <AppButton label="Review inspection" onPress={submit} />
    </ScrollView>
  );
}

function ReviewScreen({ navigation }: any) {
  const { draft, addRecord, resetDraft } = useSession();
  const timestamp = useRef(new Date().toISOString()).current;
  const rows = [
    ['Group code', GROUP_CODE],
    ['Vendor alias', draft.alias.trim()],
    ['Stall code', draft.stallCode.trim()],
    ['Category', draft.category ?? ''],
    ['Contact number', draft.phone.replace(/\s/g, '')],
    ['Risk level', draft.risk ?? ''],
    ['Consent', draft.consent ? 'Confirmed' : 'Not confirmed'],
    ['Timestamp', new Date(timestamp).toLocaleString('en-GB')],
  ];
  const save = () => {
    if (Object.keys(validate(draft)).length > 0) return navigation.goBack();
    addRecord({
      id: `rec-${Date.now()}`, alias: draft.alias.trim(), stallCode: draft.stallCode.trim(), category: draft.category,
      phone: draft.phone.replace(/\s/g, ''), risk: draft.risk, consent: draft.consent, imageUri: draft.imageUri,
      timestamp, groupCode: GROUP_CODE,
    });
    resetDraft();
    navigation.popToTop();
    navigation.navigate('RecordsTab', { screen: 'Records' });
    Alert.alert('Inspection saved', 'The record was added to this session.');
  };
  return (
    <ScrollView style={s.screen} contentContainerStyle={{ padding: space.lg, paddingBottom: 48 }}>
      <DetailRows rows={rows} />
      {draft.imageUri ? (
        <Image source={{ uri: draft.imageUri }} style={s.bigImg} accessibilityLabel="Evidence photo" />
      ) : (
        <Text style={s.noImg}>No evidence photo attached.</Text>
      )}
      <View style={s.rowWrap}>
        <AppButton label="Back to edit" onPress={() => navigation.goBack()} variant="outline" flex />
        <AppButton label="Save inspection" onPress={save} flex />
      </View>
    </ScrollView>
  );
}

            </View>
          </Pressable>
        );
      }}
    />
  );
}

function DetailsScreen({ route }: any) {
  const { records } = useSession();
  const r = records.find((x: any) => x.id === route.params.recordId);
  if (!r) return <EmptyState title="Record not found" message="Go back and choose another record." />;
  const rows = [
    ['Group code', r.groupCode],
    ['Vendor alias', r.alias],
    ['Stall code', r.stallCode],
    ['Category', r.category],
    ['Contact number', r.phone],
    ['Risk level', r.risk],
    ['Consent', r.consent ? 'Confirmed' : 'Not confirmed'],
    ['Timestamp', new Date(r.timestamp).toLocaleString('en-GB')],
  ];
  return (
    <ScrollView style={s.screen} contentContainerStyle={{ padding: space.lg, paddingBottom: 48 }}>
      <DetailRows rows={rows} />
      {r.imageUri ? (
        <Image source={{ uri: r.imageUri }} style={s.bigImg} accessibilityLabel="Evidence photo" />
      ) : (
        <Text style={s.noImg}>No evidence photo attached.</Text>
      )}
    </ScrollView>
  );
}
